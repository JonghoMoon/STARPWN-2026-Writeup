#!/bin/sh
set -eu

artifact="${1:-}"
if [ -z "$artifact" ]; then
  echo "usage: sign.sh <artifact>"
  exit 1
fi

if [ -z "${SIGNING_KEY:-}" ]; then
  echo "missing signing key"
  exit 1
fi

if [ ! -f "$artifact" ]; then
  echo "artifact not found: $artifact"
  exit 1
fi

echo "signing release artifact for ${DEPLOY_ENV:-unknown}"
echo "artifact signed"
